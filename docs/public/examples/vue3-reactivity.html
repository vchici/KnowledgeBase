本页面真实在跑的核心源码

// ── 与笔记 02 一致的最小模型（本页面只是额外加了日志）──
let activeEffect = null                 // 全局指针：当前正在运行的 effect
const deps = new Map()                  // 订阅表：key → Set<effect>

function reactive(obj) {
  return new Proxy(obj, {
    get(target, key) {
      if (activeEffect) {               // ① 依赖收集：谁在读这个 key，就把谁记入名单
        if (!deps.has(key)) deps.set(key, new Set())
        deps.get(key).add(activeEffect)
      }
      return target[key]
    },
    set(target, key, value) {
      target[key] = value
      deps.get(key)?.forEach(fn => fn()) // ② 触发更新：重跑名单里的所有 effect
      return true
    }
  })
}

function effect(tag, fn) {              // 运行期间读到的每个 key 都会被收集
  fn.tag = tag
  activeEffect = fn
  fn()
  activeEffect = null
}

function stop(fn) {                     // 扩展：把 fn 从所有名单里摘掉（真实 Vue 3 同名 API）
  for (const set of deps.values()) set.delete(fn)
}

看完应该明白的四件事

  1. 收集发生在 get：effect 运行期间读了哪些 key，就被记进哪些名单 —— 所以「渲染函数用了什么数据，改什么数据就会更新它」。
  2. 触发发生在 set：名单是 Map<key, Set<函数>>，某个 key 变了就把对应的 Set 全部重跑。
  3. Proxy 拦的是整个对象：连「定义时不存在的 key」（msg）也能拦到并触发 —— 这正是 Vue 2 的 defineProperty 做不到、需要 $set 补救的点。
  4. 只有通过 reactive 返回的代理操作才会被拦截；直接改原始对象 raw，数据变了但界面不动，还会造成数据与视图悄悄脱节。

对应笔记：docs/FullStack/前端/02-前端框架与原理.md。 真实 Vue 3 的差异：订阅表是 WeakMap<对象, Map<属性, Set>>；effect 嵌套用栈管理； 重跑前先清除旧依赖（处理 v-if 分支切换）；在此之上才有 ref / computed / watch。
